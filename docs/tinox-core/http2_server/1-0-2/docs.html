Hello server

import tinox.core.http2_server;

class Main
{
    fnc main() -> Int32
    {
        let server: Http2Server = Http2Server::new(8443);
        server.get("/hello", ctx => {
            ctx.response.text("Hello over HTTP/2!");
        });
        server.listen();
        return 0;
    }
}
