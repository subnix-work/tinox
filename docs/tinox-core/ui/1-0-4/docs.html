Annotated counter

import tinox.core.ui;
import tinox.core.websocket;
import tinox.core.http_server;

// Annotation-driven: no HttpServer/WsServer wiring anywhere in the
// program -- @TinoxUIApp generates the HTTP shell/client-JS server and
// the WebSocket accept loop, calling the class's own @View method to
// build/rebuild the component tree (full-resend rendering, automatic
// re-render after every event).
@TinoxUIApp(8080, 8081)
class CounterApp
{
    var clicks: Int64;

    @View
    fn render() -> Component
    {
        let app: CounterApp = this;
        return Component::vbox([
            Component::label("Clicks: " + this.clicks.toString()),
            Component::button("Click me", fn(v: String) { app.clicks = app.clicks + 1; })
        ]);
    }
}

Widgets and routing

import tinox.core.ui;
import tinox.core.websocket;
import tinox.core.http_server;

// A wider tour of the widget library plus client-side routing:
// Component::link(text, path, onNavigate) reports its target path back
// through the same onEvent/dispatchEvent pipeline every other
// interactive widget uses -- an app switches views by keeping its own
// current-view state (a plain instance field) and branching inside its
// own @View method, no separate routing annotation needed.
@TinoxUIApp(8080, 8081)
class ShowcaseApp
{
    var isAbout: Bool;
    var quantity: Float64;

    @View
    fn render() -> Component
    {
        let app: ShowcaseApp = this;
        let nav: Component = Component::hbox([
            Component::link("Home", "/", fn(p: String) { app.isAbout = false; }),
            Component::link("About", "/about", fn(p: String) { app.isAbout = true; })
        ]);
        if this.isAbout
        {
            return Component::vbox([nav, Component::heading("About", 1)]);
        }
        return Component::vbox([
            nav,
            Component::heading("Widgets", 1),
            Component::numberField(this.quantity, fn(v: Float64) { app.quantity = v; }),
            Component::progressBar(this.quantity)
        ]);
    }
}
