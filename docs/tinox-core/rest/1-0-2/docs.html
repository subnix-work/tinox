Client call

import tinox.core.rest.client;

class Main
{
    fnc main() -> Int32
    {
        let client: RestClient = RestClient::new("https://api.example.com");
        client.bearerAuth("my-token");

        let response: RestResponse = client.get("/users/42");
        if (response.isOk())
        {
            println("body: " + response.text());
        }
        else
        {
            println("request failed with status " + response.status().toString());
        }

        let created: RestResponse = client.post("/users", "{\"name\":\"Ada\"}");
        created.ensureSuccess();
        println("created: " + created.text());

        return 0;
    }
}

Annotated controller

import tinox.core.rest.server;
import tinox.core.json;

// Annotation-driven: no explicit main()/RestApi anywhere in the program —
// the compiler discovers @ApplicationComponent classes and generates the
// server entry point itself.
@ApplicationComponent
@Path("/api/v1")
class GreetController
{
    @GET
    @Path("/hello/:name")
    @Produces("application/json")
    fn hello(@PathParam name: String, @HttpContext ctx: HttpContext) -> HttpContext
    {
        ctx.response.json("{\"message\":\"Hello, " + name + "!\"}");
        return ctx;
    }
}
